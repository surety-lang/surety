def flatten_order(order):
    order_id = order["id"]
    email = order["customer_email"]
    out = []
    for item in order["items"]:
        out.append({
            "line_id": item["id"],
            "order_id": order_id,
            "customer_email": email,
            "amount": item["amount"],
            "item_id": item["item_id"],
        })
    return out
