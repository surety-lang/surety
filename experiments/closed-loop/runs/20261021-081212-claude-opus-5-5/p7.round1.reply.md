def enrich_orders(orders, users):
    names_by_id = {}
    for u in users:
        names_by_id[u["id"]] = u["user_name"]

    out = []
    for o in orders:
        user_id = o["user_id"]
        if user_id is not None and user_id in names_by_id:
            user_name = names_by_id[user_id]
        else:
            user_name = None
        out.append({
            "id": o["id"],
            "customer_email": o["customer_email"],
            "amount": o["amount"],
            "user_id": user_id,
            "user_name": user_name,
        })
    return out
