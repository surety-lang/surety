def well_formed_email(email):
    if not isinstance(email, str):
        return False
    if email != email.strip() or any(c.isspace() for c in email):
        return False
    if email.count("@") != 1:
        return False
    local, domain = email.split("@")
    if not local or not domain:
        return False
    if local.startswith(".") or local.endswith(".") or ".." in local:
        return False
    labels = domain.split(".")
    if len(labels) < 2:
        return False
    for label in labels:
        if not label:
            return False
        if label.startswith("-") or label.endswith("-"):
            return False
        for c in label:
            if not (c.isalnum() or c == "-"):
                return False
    return True


def _email_error():
    return {
        "code": "e-email-shape",
        "display": "Please provide a valid email",
    }


def _age_error():
    return {
        "code": "e-age-invalid",
        "display": "You are not old enough or did not enter a valid birthday.",
    }


def _password_error():
    return {
        "code": "e-pass-length",
        "display": "Your password must be at least 12 characters",
    }


def _age_ok(age):
    return 13 <= age <= 120


def _pass_ok(password):
    return len(password) >= 12


def validate_signup(p):
    errs = []
    if not well_formed_email(p["email"]):
        errs.append(_email_error())
    if not _age_ok(p["age"]):
        errs.append(_age_error())
    if not _pass_ok(p["password"]):
        errs.append(_password_error())
    if errs:
        return {"err": errs}
    return {"ok": {"email": p["email"], "age": p["age"], "password": p["password"]}}
