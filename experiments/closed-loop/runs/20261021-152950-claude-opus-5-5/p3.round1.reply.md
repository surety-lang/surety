def group_by_customer(orders):
    groups = {}
    order_keys = []
    for o in orders:
        email = o["customer_email"]
        if email not in groups:
            groups[email] = {"customer_email": email, "orders": [], "total_amount": 0}
            order_keys.append(email)
        g = groups[email]
        g["orders"].append({"amount": o["amount"], "customer_email": email})
        g["total_amount"] += o["amount"]
    return [groups[k] for k in order_keys]
