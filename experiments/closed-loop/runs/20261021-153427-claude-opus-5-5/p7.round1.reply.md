def enrich_orders(orders, users):
    names = {}
    for u in users:
        names[u["id"]] = u["user_name"]
    out = []
    for o in orders:
        uid = o["user_id"]
        out.append({
            "id": o["id"],
            "customer_email": o["customer_email"],
            "amount": o["amount"],
            "user_id": uid,
            "user_name": names.get(uid) if uid is not None else None,
        })
    return out
